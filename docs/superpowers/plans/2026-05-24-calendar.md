```ts
// utilities/calendar.utility.ts
import ICAL from "ical.js";
import type {
  CalendarEventItem,
  CalendarParseOptions,
  CalendarParseResult,
  CalendarSortOrder,
} from "../models/calendar.model";

export function parseCalendarEvents(
  calendarText: string,
  options: CalendarParseOptions,
  now: Date = new Date(),
): CalendarParseResult {
  const warnings: string[] = [];
  const stats = { totalEvents: 0, filteredEvents: 0, recurringExpanded: 0, canceledExcluded: 0, timezoneWarnings: 0 };

  let jcal: unknown;
  try {
    jcal = ICAL.parse(calendarText);
  } catch (err) {
    warnings.push(`Failed to parse calendar: ${String(err)}`);
    return { events: [], warnings, stats };
  }

  const comp = new ICAL.Component(jcal);
  const windowStart = options.includePastEvents
    ? new Date(now.getTime() - options.windowDays * 86400000)
    : now;
  const windowEnd = new Date(now.getTime() + options.windowDays * 86400000);

  const allEvents: CalendarEventItem[] = [];
  const vevents = comp.getAllSubcomponents("vevent");

  for (const vevent of vevents) {
    const event = new ICAL.Event(vevent);
    stats.totalEvents++;

    if (options.expandRecurringEvents && event.isRecurring()) {
      const iter = event.iterator();
      let next = iter.next();
      let instanceCount = 0;
      while (next && instanceCount < options.maxEvents) {
        const occurrence = event.getOccurrenceDetails(next);
        const item = icalEventToItem(occurrence.item, vevent, warnings, stats);
        if (item && isInWindow(item.start, windowStart, windowEnd)) {
          allEvents.push(item);
        }
        next = iter.next();
        instanceCount++;
        if (item?.start && item.start > windowEnd) break;
      }
      stats.recurringExpanded++;
    } else {
      const item = icalEventToItem(event, vevent, warnings, stats);
      if (item) allEvents.push(item);
    }
  }

  const filtered = allEvents.filter((item) => {
    if (!isInWindow(item.start, windowStart, windowEnd)) {
      stats.filteredEvents++;
      return false;
    }
    if (!options.includeCanceled && item.status === "CANCELLED") {
      stats.canceledExcluded++;
      return false;
    }
    if (options.filters) {
      if (!passesFilters(item, options.filters)) {
        stats.filteredEvents++;
        return false;
      }
    }
    return true;
  });

  const sorted = sortEvents(filtered, options.sortOrder);
  return { events: sorted.slice(0, options.maxEvents), warnings, stats };
}

function icalEventToItem(
  event: ICAL.Event,
  _vevent: ICAL.Component,
  warnings: string[],
  _stats: { timezoneWarnings: number },
): CalendarEventItem | null {
  const uid = event.uid;
  if (!uid) {
    warnings.push("Event is missing UID; skipping");
    return null;
  }

  const summary = event.summary || "Untitled Event";
  const description = event.description || undefined;
  const location = event.location || undefined;

  let start: Date | undefined;
  let end: Date | undefined;
  try {
    start = event.startDate?.toJSDate();
    end = event.endDate?.toJSDate();
  } catch {}

  let created: Date | undefined;
  let lastModified: Date | undefined;
  try {
    created = (event.component.getFirstPropertyValue("created") as ICAL.Time | null)?.toJSDate();
    lastModified = (event.component.getFirstPropertyValue("last-modified") as ICAL.Time | null)?.toJSDate();
  } catch {}

  const urlProp = event.component.getFirstPropertyValue("url");
  const url = urlProp ? String(urlProp) : undefined;

  const organizerProp = event.component.getFirstProperty("organizer");
  const organizer = organizerProp
    ? organizerProp.getParameter("cn") ?? String(organizerProp.getFirstValue())
    : undefined;

  const categoriesProp = event.component.getAllProperties("categories");
  const categories: string[] = [];
  for (const cat of categoriesProp) {
    const values = cat.getValues();
    for (const v of values) {
      if (typeof v === "string") categories.push(v);
      else if (Array.isArray(v)) categories.push(...v.map(String));
    }
  }

  const statusProp = event.component.getFirstPropertyValue("status");
  const status = statusProp ? String(statusProp) : undefined;

  const recurrenceId = event.recurrenceId
    ? event.recurrenceId.toJSDate().toISOString()
    : undefined;

  return { uid, recurrenceId, summary, description, location, url, organizer, categories, status, start, end, created, lastModified };
}

function isInWindow(date: Date | undefined, start: Date, end: Date): boolean {
  if (!date) return true;
  return date >= start && date <= end;
}

function sortEvents(events: CalendarEventItem[], order: CalendarSortOrder): CalendarEventItem[] {
  const copy = [...events];
  switch (order) {
    case "startAsc":
      return copy.sort((a, b) => (a.start?.getTime() ?? 0) - (b.start?.getTime() ?? 0));
    case "startDesc":
      return copy.sort((a, b) => (b.start?.getTime() ?? 0) - (a.start?.getTime() ?? 0));
    case "modifiedDesc":
      return copy.sort((a, b) => (b.lastModified?.getTime() ?? 0) - (a.lastModified?.getTime() ?? 0));
    default:
      return copy;
  }
}

function passesFilters(
  item: CalendarEventItem,
  filters: NonNullable<CalendarParseOptions["filters"]>,
): boolean {
  const check = (rule: { field: string; type: string; value: string; caseSensitive?: boolean }, include: boolean): boolean => {
    const fieldValue = String((item as Record<string, unknown>)[rule.field] ?? "");
    const matches = rule.type === "keyword"
      ? (rule.caseSensitive ? fieldValue : fieldValue.toLowerCase()).includes(
          rule.caseSensitive ? rule.value : rule.value.toLowerCase()
        )
      : new RegExp(rule.value, rule.caseSensitive ? "" : "i").test(fieldValue);
    return include ? matches : !matches;
  };

  if (filters.include?.length) {
    if (!filters.include.every((r) => check(r, true))) return false;
  }
  if (filters.exclude?.length) {
    if (filters.exclude.some((r) => !check(r, false))) return false;
  }
  return true;
}
```

```ts
import type { CalendarEventItem } from "../models/calendar.model";
import type { FeedConfig } from "../models/feed-config.model";

export function buildRSSFromCalendarData(
  events: CalendarEventItem[],
  config: FeedConfig,
): string {
  const feedName = config.feedName ?? "Calendar Feed";
  const calendarConfig = config.calendar!;

  const items = events.map((event) => {
    const title = escapeXml(event.summary || "Untitled Event");
    const guid = buildCalendarGuid(event);
    const description = buildCalendarDescription(event);
    const link = resolveCalendarLink(event, calendarConfig.url, calendarConfig.linkStrategy);
    const pubDate = getCalendarItemDate(event, calendarConfig.dateStrategy)?.toUTCString() ?? "";
    const author = event.organizer ? escapeXml(event.organizer) : "";
    const categories = event.categories.map((c) => `<category>${escapeXml(c)}</category>`).join("");

    return `
    <item>
      <title>${title}</title>
      <guid isPermaLink="false">${escapeXml(guid)}</guid>
      <description><![CDATA[${description}]]></description>
      ${link ? `<link>${escapeXml(link)}</link>` : ""}
      ${pubDate ? `<pubDate>${pubDate}</pubDate>` : ""}
      ${author ? `<author>${author}</author>` : ""}
      ${categories}
    </item>`;
  });

  return `<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0">
  <channel>
    <title>${escapeXml(feedName)}</title>
    <link>${escapeXml(calendarConfig.url)}</link>
    <description>Calendar feed: ${escapeXml(feedName)}</description>
    ${items.join("")}
  </channel>
</rss>`;
}

function buildCalendarGuid(event: CalendarEventItem): string {
  const recurrencePart = event.recurrenceId ?? event.start?.toISOString() ?? "";
  return [event.uid, recurrencePart].filter(Boolean).join("#");
}

function buildCalendarDescription(event: CalendarEventItem): string {
  const parts: string[] = [];
  if (event.start) parts.push(`Start: ${formatCalendarDate(event.start)}`);
  if (event.end) parts.push(`End: ${formatCalendarDate(event.end)}`);
  if (event.location) parts.push(`Location: ${event.location}`);
  if (event.description) parts.push("", event.description);
  return parts.join("\n");
}

function resolveCalendarLink(
  event: CalendarEventItem,
  calendarUrl: string,
  strategy: "eventUrl" | "location" | "calendarUrl" | "none",
): string | undefined {
  switch (strategy) {
    case "eventUrl":
      return event.url ?? calendarUrl;
    case "location":
      return event.location?.startsWith("http") ? event.location : calendarUrl;
    case "calendarUrl":
      return calendarUrl;
    case "none":
      return undefined;
  }
}

function getCalendarItemDate(
  event: CalendarEventItem,
  strategy: "start" | "end" | "created" | "lastModified",
): Date | undefined {
  switch (strategy) {
    case "start": return event.start;
    case "end": return event.end;
    case "created": return event.created;
    case "lastModified": return event.lastModified;
  }
}

function formatCalendarDate(date: Date): string {
  return date.toLocaleString("en-US", { weekday: "short", year: "numeric", month: "short", day: "numeric", hour: "2-digit", minute: "2-digit", timeZoneName: "short" });
}

function escapeXml(str: string): string {
  return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}
```
