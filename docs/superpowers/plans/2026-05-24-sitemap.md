```ts
// utilities/sitemap.utility.ts
import { XMLParser } from "fast-xml-parser";
import type {
  SitemapEntry,
  SitemapIndexEntry,
  SitemapParseResult,
  SitemapSortOrder,
  SitemapFeedConfig,
} from "../models/sitemap.model";

export type SitemapParseOptions = {
  maxUrlsToScan?: number;
  filters?: SitemapFeedConfig["filters"];
};

const SITEMAP_PROBE_PATHS = [
  "/sitemap.xml",
  "/sitemap_index.xml",
  "/sitemap-index.xml",
  "/wp-sitemap.xml",
  "/post-sitemap.xml",
  "/page-sitemap.xml",
];

const parser = new XMLParser({
  ignoreAttributes: false,
  parseAttributeValue: true,
  allowBooleanAttributes: true,
});

export function parseSitemapXml(
  xml: string,
  sourceUrl: string,
  now: string = new Date().toISOString(),
): SitemapParseResult {
  const warnings: string[] = [];
  const entries: SitemapEntry[] = [];
  const childSitemaps: SitemapIndexEntry[] = [];

  let parsed: Record<string, unknown>;
  try {
    parsed = parser.parse(xml) as Record<string, unknown>;
  } catch {
    warnings.push("Sitemap XML could not be parsed");
    return {
      type: "urlset",
      entries: [],
      childSitemaps: [],
      warnings,
      stats: { totalUrls: 0, urlsAfterFilters: 0, totalChildSitemaps: 0, fetchedChildSitemaps: 0, failedChildSitemaps: 0, duplicateUrls: 0 },
    };
  }

  if (parsed.sitemapindex) {
    const index = parsed.sitemapindex as Record<string, unknown>;
    const sitemaps = toArray(index.sitemap);
    for (const s of sitemaps) {
      const sm = s as Record<string, unknown>;
      if (sm.loc) {
        childSitemaps.push({ loc: String(sm.loc), lastmod: sm.lastmod ? String(sm.lastmod) : undefined, sourceSitemapUrl: sourceUrl });
      }
    }
    return {
      type: "sitemapindex",
      entries: [],
      childSitemaps,
      warnings,
      stats: { totalUrls: 0, urlsAfterFilters: 0, totalChildSitemaps: childSitemaps.length, fetchedChildSitemaps: 0, failedChildSitemaps: 0, duplicateUrls: 0 },
    };
  }

  const urlset = parsed.urlset as Record<string, unknown> | undefined;
  if (!urlset) {
    warnings.push("Sitemap XML does not contain a urlset or sitemapindex element");
    return {
      type: "urlset",
      entries: [],
      childSitemaps: [],
      warnings,
      stats: { totalUrls: 0, urlsAfterFilters: 0, totalChildSitemaps: 0, fetchedChildSitemaps: 0, failedChildSitemaps: 0, duplicateUrls: 0 },
    };
  }

  const urls = toArray(urlset.url);
  const seen = new Map<string, SitemapEntry>();
  let order = 0;
  let duplicateCount = 0;

  for (const u of urls) {
    const url = u as Record<string, unknown>;
    if (!url.loc) continue;
    const loc = String(url.loc).trim();
    const entry: SitemapEntry = {
      loc,
      lastmod: url.lastmod ? String(url.lastmod).trim() : undefined,
      changefreq: url.changefreq ? String(url.changefreq).trim() : undefined,
      priority: url.priority != null ? Number(url.priority) : undefined,
      sourceSitemapUrl: sourceUrl,
      discoveredAt: now,
      order: order++,
    };
    if (seen.has(loc)) {
      duplicateCount++;
    }
    seen.set(loc, entry);
  }

  const deduped = Array.from(seen.values());
  const hasLastmod = deduped.some((e) => e.lastmod);
  if (!hasLastmod && deduped.length > 0) {
    warnings.push("No <lastmod> values found in sitemap; using first-seen date as fallback");
  }

  return {
    type: "urlset",
    entries: deduped,
    childSitemaps: [],
    warnings,
    stats: {
      totalUrls: deduped.length,
      urlsAfterFilters: deduped.length,
      totalChildSitemaps: 0,
      fetchedChildSitemaps: 0,
      failedChildSitemaps: 0,
      duplicateUrls: duplicateCount,
    },
  };
}

export function applySitemapFilters(
  entries: SitemapEntry[],
  filters?: SitemapFeedConfig["filters"],
): SitemapEntry[] {
  if (!filters) return entries;
  let result = entries;

  if (filters.include && filters.include.length > 0) {
    result = result.filter((entry) =>
      filters.include!.some((rule) => matchesRule(entry, rule))
    );
  }

  if (filters.exclude && filters.exclude.length > 0) {
    result = result.filter(
      (entry) => !filters.exclude!.some((rule) => matchesRule(entry, rule))
    );
  }

  return result;
}

function matchesRule(entry: SitemapEntry, rule: { type: "keyword" | "regex"; field: "loc" | "lastmod" | "changefreq" | "priority"; value: string; caseSensitive?: boolean }): boolean {
  const fieldValue = String(entry[rule.field] ?? "");
  if (rule.type === "keyword") {
    const value = rule.caseSensitive ? rule.value : rule.value.toLowerCase();
    const target = rule.caseSensitive ? fieldValue : fieldValue.toLowerCase();
    return target.includes(value);
  }
  const flags = rule.caseSensitive ? "" : "i";
  return new RegExp(rule.value, flags).test(fieldValue);
}

export function sortSitemapEntries(
  entries: SitemapEntry[],
  sortOrder: SitemapSortOrder,
): SitemapEntry[] {
  const copy = [...entries];
  switch (sortOrder) {
    case "lastmodDesc":
      return copy.sort((a, b) => (b.lastmod ?? "").localeCompare(a.lastmod ?? ""));
    case "lastmodAsc":
      return copy.sort((a, b) => (a.lastmod ?? "").localeCompare(b.lastmod ?? ""));
    case "firstSeenDesc":
      return copy.sort((a, b) => b.discoveredAt.localeCompare(a.discoveredAt));
    case "urlAsc":
      return copy.sort((a, b) => a.loc.localeCompare(b.loc));
    case "sitemapOrder":
    default:
      return copy.sort((a, b) => a.order - b.order);
  }
}

export function discoverSitemapsFromRobotsTxt(robotsTxt: string): string[] {
  const urls: string[] = [];
  for (const line of robotsTxt.split("\n")) {
    const match = /^Sitemap:\s*(.+)$/i.exec(line.trim());
    if (match) {
      urls.push(match[1].trim());
    }
  }
  return urls;
}

export async function discoverSitemaps(siteUrl: string): Promise<string[]> {
  const base = new URL(siteUrl).origin;
  const discovered: string[] = [];

  try {
    const robotsRes = await fetch(`${base}/robots.txt`);
    if (robotsRes.ok) {
      const text = await robotsRes.text();
      discovered.push(...discoverSitemapsFromRobotsTxt(text));
    }
  } catch {}

  if (discovered.length === 0) {
    for (const path of SITEMAP_PROBE_PATHS) {
      try {
        const res = await fetch(`${base}${path}`, { method: "HEAD" });
        if (res.ok) {
          discovered.push(`${base}${path}`);
        }
      } catch {}
    }
  }

  return [...new Set(discovered)];
}

export async function fetchAndParseSitemap(
  sitemapUrl: string,
  options: SitemapParseOptions = {},
): Promise<SitemapParseResult> {
  const now = new Date().toISOString();
  const { maxUrlsToScan = 10000 } = options;

  let xml: string;
  try {
    const res = await fetch(sitemapUrl);
    if (!res.ok) {
      return {
        type: "urlset",
        entries: [],
        childSitemaps: [],
        warnings: [`Sitemap fetch failed: HTTP ${res.status}`],
        stats: { totalUrls: 0, urlsAfterFilters: 0, totalChildSitemaps: 0, fetchedChildSitemaps: 0, failedChildSitemaps: 0, duplicateUrls: 0 },
      };
    }
    const contentType = res.headers.get("content-type") ?? "";
    if (contentType.includes("gzip") || sitemapUrl.endsWith(".gz")) {
      const buffer = await res.arrayBuffer();
      const { gunzipSync } = await import("zlib");
      xml = gunzipSync(Buffer.from(buffer)).toString("utf8");
    } else {
      xml = await res.text();
    }
  } catch (err) {
    return {
      type: "urlset",
      entries: [],
      childSitemaps: [],
      warnings: [`Sitemap fetch error: ${String(err)}`],
      stats: { totalUrls: 0, urlsAfterFilters: 0, totalChildSitemaps: 0, fetchedChildSitemaps: 0, failedChildSitemaps: 0, duplicateUrls: 0 },
    };
  }

  const parsed = parseSitemapXml(xml, sitemapUrl, now);

  if (parsed.type === "sitemapindex") {
    const allEntries: SitemapEntry[] = [];
    let fetched = 0;
    let failed = 0;
    for (const child of parsed.childSitemaps) {
      if (allEntries.length >= maxUrlsToScan) break;
      try {
        const childResult = await fetchAndParseSitemap(child.loc, { maxUrlsToScan: maxUrlsToScan - allEntries.length });
        allEntries.push(...childResult.entries);
        fetched++;
        parsed.warnings.push(...childResult.warnings);
      } catch {
        failed++;
        parsed.warnings.push(`Failed to fetch child sitemap: ${child.loc}`);
      }
    }
    return {
      ...parsed,
      entries: allEntries.slice(0, maxUrlsToScan),
      stats: {
        ...parsed.stats,
        totalUrls: allEntries.length,
        urlsAfterFilters: allEntries.length,
        fetchedChildSitemaps: fetched,
        failedChildSitemaps: failed,
      },
    };
  }

  const filtered = options.filters
    ? applySitemapFilters(parsed.entries, options.filters)
    : parsed.entries;

  return {
    ...parsed,
    entries: filtered.slice(0, maxUrlsToScan),
    stats: {
      ...parsed.stats,
      urlsAfterFilters: filtered.length,
    },
  };
}

function toArray(value: unknown): unknown[] {
  if (!value) return [];
  return Array.isArray(value) ? value : [value];
}
```

```ts
// utilities/sitemap-json-ld.utility.ts
import type {
  SitemapEntry,
  SitemapFeedConfig,
  SitemapJsonLdAnalysisResult,
  SitemapJsonLdSample,
  PageMetadata,
} from "../models/sitemap.model";
import type { JsonLdFeedFieldMapping } from "../models/json-ld.model";
import { extractPageMetadata } from "./page-metadata.utility";

type FallbackConfig = {
  enabled: boolean;
  order: ReadonlyArray<"openGraph" | "htmlMeta" | "sitemap" | "url">;
};

export type SitemapFeedItem = {
  title?: string;
  description?: string;
  link?: string;
  pubDate?: string;
  author?: string;
  enclosure?: string;
  guid?: string;
  categories?: string;
};

export function buildFeedItemFromSitemapEntryJsonLd(
  html: string,
  entry: SitemapEntry,
  mapping: JsonLdFeedFieldMapping,
  fallback: FallbackConfig | undefined,
): SitemapFeedItem {
  const jsonLd = extractJsonLdFromHtml(html);
  const pageMeta = extractPageMetadata(html, entry.loc);

  const item: SitemapFeedItem = {};

  const fields = ["title", "description", "link", "pubDate", "author", "enclosure", "guid", "categories"] as const;
  for (const field of fields) {
    const path = (mapping as Record<string, string | undefined>)[field];
    let value: string | undefined;

    if (path && jsonLd) {
      value = getNestedPath(jsonLd, path);
    }

    if (!value && fallback?.enabled) {
      value = applyFallback(field, entry, pageMeta, fallback.order);
    }

    if (value) {
      (item as Record<string, string>)[field] = value;
    }
  }

  return item;
}

function applyFallback(
  field: string,
  entry: SitemapEntry,
  pageMeta: PageMetadata,
  order: ReadonlyArray<"openGraph" | "htmlMeta" | "sitemap" | "url">,
): string | undefined {
  for (const source of order) {
    const value = getFallbackValue(field, source, entry, pageMeta);
    if (value) return value;
  }
  return undefined;
}

function getFallbackValue(
  field: string,
  source: "openGraph" | "htmlMeta" | "sitemap" | "url",
  entry: SitemapEntry,
  pageMeta: PageMetadata,
): string | undefined {
  switch (source) {
    case "openGraph":
      if (field === "title") return pageMeta.title;
      if (field === "description") return pageMeta.description;
      if (field === "enclosure") return pageMeta.image;
      if (field === "link") return pageMeta.canonicalUrl;
      return undefined;
    case "htmlMeta":
      if (field === "description") return pageMeta.description;
      return undefined;
    case "sitemap":
      if (field === "pubDate") return entry.lastmod;
      if (field === "link") return entry.loc;
      if (field === "guid") return entry.loc;
      return undefined;
    case "url":
      if (field === "link") return entry.loc;
      if (field === "guid") return entry.loc;
      if (field === "title") {
        const path = new URL(entry.loc).pathname;
        return path.split("/").filter(Boolean).at(-1) ?? entry.loc;
      }
      return undefined;
  }
}

function extractJsonLdFromHtml(html: string): Record<string, unknown> | null {
  const match = /<script[^>]*type=["']application\/ld\+json["'][^>]*>([\s\S]*?)<\/script>/i.exec(html);
  if (!match) return null;
  try {
    return JSON.parse(match[1]) as Record<string, unknown>;
  } catch {
    return null;
  }
}

function getNestedPath(obj: Record<string, unknown>, path: string): string | undefined {
  if (path === "@id") {
    const id = obj["@id"];
    return id ? String(id) : undefined;
  }
  const parts = path.split(".");
  let current: unknown = obj;
  for (const part of parts) {
    if (!current || typeof current !== "object") return undefined;
    current = (current as Record<string, unknown>)[part];
  }
  return current ? String(current) : undefined;
}

export function compareSitemapJsonLdSamples(
  samples: SitemapJsonLdSample[],
): JsonLdFeedFieldMapping {
  const successfulSamples = samples.filter((s) => s.success && s.jsonLdFound);
  if (successfulSamples.length === 0) return {};

  const pathFrequency = new Map<string, number>();
  for (const sample of successfulSamples) {
    for (const path of sample.availablePaths) {
      pathFrequency.set(path.path, (pathFrequency.get(path.path) ?? 0) + 1);
    }
  }

  const mapping: JsonLdFeedFieldMapping = {};
  const fieldCandidates: Record<string, string[]> = {
    title: ["headline", "name", "title"],
    description: ["description", "abstract", "summary"],
    link: ["url", "mainEntityOfPage", "@id"],
    pubDate: ["datePublished", "dateCreated", "startDate"],
    author: ["author.name", "creator.name"],
    enclosure: ["image.url", "image.0.url", "thumbnailUrl"],
    guid: ["@id", "url"],
    categories: ["articleSection", "genre", "keywords.0"],
  };

  for (const [field, candidates] of Object.entries(fieldCandidates)) {
    for (const candidate of candidates) {
      if ((pathFrequency.get(candidate) ?? 0) > 0) {
        (mapping as Record<string, string>)[field] = candidate;
        break;
      }
    }
  }

  return mapping;
}
```

```ts
// utilities/sitemap-state.utility.ts
import { readFileSync, writeFileSync, mkdirSync, existsSync } from "fs";
import { join } from "path";
import type { SitemapFeedState } from "../models/sitemap.model";

const STATE_DIR = "./feed-state/sitemaps";

export function loadSitemapState(feedId: string): SitemapFeedState {
  const path = join(STATE_DIR, `${feedId}.json`);
  if (!existsSync(path)) return { urls: {} };
  try {
    return JSON.parse(readFileSync(path, "utf8")) as SitemapFeedState;
  } catch {
    return { urls: {} };
  }
}

export function saveSitemapState(feedId: string, state: SitemapFeedState): void {
  if (!existsSync(STATE_DIR)) {
    mkdirSync(STATE_DIR, { recursive: true });
  }
  writeFileSync(join(STATE_DIR, `${feedId}.json`), JSON.stringify(state, null, 2));
}

export function updateSitemapState(
  existing: SitemapFeedState,
  currentLocs: string[],
  now: string,
): { state: SitemapFeedState; newUrls: number; modifiedUrls: number; disappearedUrls: number } {
  const updated: SitemapFeedState = { urls: { ...existing.urls } };
  let newUrls = 0;
  let modifiedUrls = 0;
  const currentSet = new Set(currentLocs);

  for (const loc of currentLocs) {
    if (!updated.urls[loc]) {
      updated.urls[loc] = { firstSeenAt: now, lastSeenAt: now };
      newUrls++;
    } else {
      updated.urls[loc] = { ...updated.urls[loc], lastSeenAt: now };
      modifiedUrls++;
    }
  }

  const disappearedUrls = Object.keys(existing.urls).filter((loc) => !currentSet.has(loc)).length;
  return { state: updated, newUrls, modifiedUrls, disappearedUrls };
}
```

```ts
// utilities/sitemap-builder.utility.ts
import type { SitemapEntry, SitemapFeedConfig, SitemapFeedState } from "../models/sitemap.model";
import type { FeedConfig } from "../models/feed-config.model";

export type SitemapFeed = {
  title: string;
  link: string;
  description: string;
  items: SitemapFeedItem[];
};

export type SitemapFeedItem = {
  id: string;
  title: string;
  link: string;
  date?: Date;
  description?: string;
  author?: string;
  enclosure?: { url: string; type: string };
};

export function titleFromSitemapUrl(
  loc: string,
  strategy: SitemapFeedConfig["titleStrategy"],
): string {
  try {
    const url = new URL(loc);
    switch (strategy) {
      case "path":
        return url.pathname.split("/").filter(Boolean).at(-1) ?? url.pathname;
      case "url":
        return loc;
      case "hostnameAndPath":
        return url.hostname + url.pathname;
      case "bestAvailable":
      default:
        return url.pathname.split("/").filter(Boolean).at(-1) ?? url.hostname;
    }
  } catch {
    return loc;
  }
}

export function buildFeedFromSitemapEntries(
  entries: SitemapEntry[],
  feedConfig: FeedConfig,
  sitemapConfig: SitemapFeedConfig,
  state?: SitemapFeedState,
): SitemapFeed {
  const items: SitemapFeedItem[] = entries.map((entry) => {
    const title = titleFromSitemapUrl(entry.loc, sitemapConfig.titleStrategy);
    const date = resolveDateForEntry(entry, sitemapConfig.dateStrategy, state);

    return {
      id: entry.loc,
      title,
      link: entry.loc,
      date,
      description: buildSitemapDescription(entry),
    };
  });

  return {
    title: (feedConfig as any).feedName ?? "Sitemap Feed",
    link: sitemapConfig.url,
    description: `Feed generated from sitemap: ${sitemapConfig.url}`,
    items,
  };
}

function resolveDateForEntry(
  entry: SitemapEntry,
  strategy: SitemapFeedConfig["dateStrategy"],
  state?: SitemapFeedState,
): Date | undefined {
  const firstSeen = state?.urls[entry.loc]?.firstSeenAt ?? entry.discoveredAt;

  switch (strategy) {
    case "lastmodOnly":
      return entry.lastmod ? new Date(entry.lastmod) : undefined;
    case "lastmodOrFirstSeen":
      return entry.lastmod ? new Date(entry.lastmod) : new Date(firstSeen);
    case "firstSeen":
      return new Date(firstSeen);
    case "currentRun":
      return new Date();
    case "jsonLdOrLastmodOrFirstSeen":
    case "bestAvailable":
    default:
      return entry.lastmod ? new Date(entry.lastmod) : new Date(firstSeen);
  }
}

function buildSitemapDescription(entry: SitemapEntry): string {
  const parts: string[] = ["Discovered from sitemap."];
  if (entry.lastmod) parts.push(`Last modified: ${entry.lastmod}.`);
  if (entry.changefreq) parts.push(`Change frequency: ${entry.changefreq}.`);
  return parts.join(" ");
}
```

```ts
// routes/sitemap.ts
import { Hono } from "hono";
import { fetchAndParseSitemap, applySitemapFilters, sortSitemapEntries } from "../utilities/sitemap.utility";
import { buildFeedFromSitemapEntries } from "../utilities/sitemap-builder.utility";
import type { FeedConfig } from "../models/feed-config.model";
import type { SitemapEntryPreview, SitemapPreviewResponse } from "../models/sitemap.model";

const router = new Hono();

router.post("/preview/sitemap", async (c) => {
  const body = await c.req.json<{ config: FeedConfig }>();
  const { config } = body;

  if (!config.sitemap) {
    return c.json({ error: "Missing sitemap config" }, 400);
  }

  const sitemapConfig = config.sitemap;

  const parseResult = await fetchAndParseSitemap(sitemapConfig.url, {
    maxUrlsToScan: sitemapConfig.maxUrlsToScan,
    filters: sitemapConfig.filters,
  });

  const sorted = sortSitemapEntries(
    applySitemapFilters(parseResult.entries, sitemapConfig.filters),
    sitemapConfig.sortOrder,
  );

  const previewEntries: SitemapEntryPreview[] = sorted.slice(0, Math.min(sitemapConfig.maxItems, 50)).map((entry) => ({
    loc: entry.loc,
    lastmod: entry.lastmod,
    included: true,
    outputItem: {
      title: entry.loc.split("/").filter(Boolean).at(-1) ?? entry.loc,
      link: entry.loc,
      pubDate: entry.lastmod,
      guid: entry.loc,
    },
  }));

  const feed = buildFeedFromSitemapEntries(sorted.slice(0, sitemapConfig.maxItems), config, sitemapConfig);

  const response: SitemapPreviewResponse = {
    entries: previewEntries,
    warnings: parseResult.warnings,
    stats: parseResult.stats,
  };

  return c.json(response);
});

export default router;
```

The form has these sections (detailed wireframe in the spec):

- "Analyze Sitemap" calls `POST /preview/sitemap` and shows the preview table
- Feed mode `jsonLd`/`jsonLdWithFallback` reveals the JSON-LD section (reuse JSON-LD mapping editor from JSON-LD Integration components)
- `changeDetection` option shows as disabled with "coming later" label
- Preview table columns: Last Modified, URL, Included, Output Title

```tsx
// frontend/src/components/forms/SitemapForm.tsx
import React, { useState } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import type { SitemapFeedConfig, SitemapPreviewResponse } from "@/models/sitemap.model";

interface SitemapFormProps {
  value: Partial<SitemapFeedConfig>;
  onChange: (config: Partial<SitemapFeedConfig>) => void;
  feedName: string;
  onFeedNameChange: (name: string) => void;
  onSave: () => void;
}

export function SitemapForm({ value, onChange, feedName, onFeedNameChange, onSave }: SitemapFormProps) {
  const [preview, setPreview] = useState<SitemapPreviewResponse | null>(null);
  const [analyzing, setAnalyzing] = useState(false);

  const config: SitemapFeedConfig = {
    inputMode: "exact",
    url: "",
    mode: "urlList",
    maxItems: 50,
    maxUrlsToScan: 500,
    sortOrder: "lastmodDesc",
    dateStrategy: "lastmodOrFirstSeen",
    titleStrategy: "path",
    descriptionStrategy: "sitemapMetadata",
    ...value,
  };

  async function handleAnalyze() {
    if (!config.url) return;
    setAnalyzing(true);
    try {
      const res = await fetch("/preview/sitemap", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ config: { feedType: "sitemap", feedName, sitemap: config } }),
      });
      const data = await res.json();
      setPreview(data);
    } finally {
      setAnalyzing(false);
    }
  }

  return (
    <div className="space-y-6">
      {/* Sitemap Source */}
      <div className="space-y-2">
        <Label>Sitemap source</Label>
        <RadioGroup
          value={config.inputMode}
          onValueChange={(v) => onChange({ ...config, inputMode: v as "exact" | "discover" })}
        >
          <div className="flex items-center space-x-2">
            <RadioGroupItem value="exact" id="exact" />
            <Label htmlFor="exact">Exact sitemap URL</Label>
          </div>
          <div className="flex items-center space-x-2">
            <RadioGroupItem value="discover" id="discover" />
            <Label htmlFor="discover">Discover from website URL</Label>
          </div>
        </RadioGroup>
        <Input
          placeholder="https://example.com/sitemap.xml"
          value={config.url}
          onChange={(e) => onChange({ ...config, url: e.target.value })}
        />
        <Button onClick={handleAnalyze} disabled={analyzing || !config.url}>
          {analyzing ? "Analyzing..." : "Analyze Sitemap"}
        </Button>
      </div>

      {/* Feed Mode */}
      <div className="space-y-2">
        <Label>What should this feed contain?</Label>
        <RadioGroup
          value={config.mode}
          onValueChange={(v) => onChange({ ...config, mode: v as SitemapFeedConfig["mode"] })}
        >
          {[
            { value: "urlList", label: "URLs from sitemap" },
            { value: "pageMetadata", label: "Enriched page metadata" },
            { value: "jsonLd", label: "Sitemap Drill Chain + JSON-LD" },
            { value: "jsonLdWithFallback", label: "Sitemap Drill Chain + JSON-LD with fallback" },
            { value: "changeDetection", label: "Page content changes (coming later)", disabled: true },
          ].map(({ value, label, disabled }) => (
            <div key={value} className="flex items-center space-x-2">
              <RadioGroupItem value={value} id={value} disabled={disabled} />
              <Label htmlFor={value} className={disabled ? "text-muted-foreground" : ""}>{label}</Label>
            </div>
          ))}
        </RadioGroup>
      </div>

      {/* Sorting & Limits */}
      <div className="grid grid-cols-3 gap-4">
        <div className="space-y-1">
          <Label>Sort by</Label>
          <Select value={config.sortOrder} onValueChange={(v) => onChange({ ...config, sortOrder: v as SitemapFeedConfig["sortOrder"] })}>
            <SelectTrigger><SelectValue /></SelectTrigger>
            <SelectContent>
              <SelectItem value="lastmodDesc">Last modified (newest first)</SelectItem>
              <SelectItem value="lastmodAsc">Last modified (oldest first)</SelectItem>
              <SelectItem value="urlAsc">URL (alphabetical)</SelectItem>
              <SelectItem value="sitemapOrder">Sitemap order</SelectItem>
            </SelectContent>
          </Select>
        </div>
        <div className="space-y-1">
          <Label>Max feed items</Label>
          <Input type="number" value={config.maxItems} onChange={(e) => onChange({ ...config, maxItems: Number(e.target.value) })} min={1} max={1000} />
        </div>
        <div className="space-y-1">
          <Label>Max URLs to scan</Label>
          <Input type="number" value={config.maxUrlsToScan} onChange={(e) => onChange({ ...config, maxUrlsToScan: Number(e.target.value) })} min={1} max={10000} />
        </div>
      </div>

      {/* Preview */}
      {preview && (
        <div className="space-y-2">
          <Label>Preview ({preview.stats.urlsAfterFilters} matched)</Label>
          {preview.warnings.length > 0 && (
            <div className="text-sm text-yellow-600">{preview.warnings.join("; ")}</div>
          )}
          <div className="border rounded overflow-auto max-h-64">
            <table className="w-full text-sm">
              <thead className="bg-muted">
                <tr>
                  <th className="px-3 py-2 text-left">Last Modified</th>
                  <th className="px-3 py-2 text-left">URL</th>
                  <th className="px-3 py-2 text-left">Output Title</th>
                </tr>
              </thead>
              <tbody>
                {preview.entries.map((entry) => (
                  <tr key={entry.loc} className="border-t">
                    <td className="px-3 py-1 text-muted-foreground">{entry.lastmod ?? "—"}</td>
                    <td className="px-3 py-1 font-mono text-xs truncate max-w-[300px]">{entry.loc}</td>
                    <td className="px-3 py-1">{entry.outputItem?.title ?? "—"}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )}

      {/* Feed name & Save */}
      <div className="space-y-2">
        <Label>Feed name</Label>
        <Input value={feedName} onChange={(e) => onFeedNameChange(e.target.value)} placeholder="my-sitemap-feed" />
        <Button onClick={onSave}>Save feed</Button>
      </div>
    </div>
  );
}
```
