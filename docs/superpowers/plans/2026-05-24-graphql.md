```ts
// utilities/structured-feed.utility.ts
import type { JsonArrayPathCandidate } from "../models/graphql.model";

const HIGH_CONFIDENCE_PATH_KEYWORDS = ["nodes", "edges", "items", "results", "data"];
const HIGH_CONFIDENCE_FIELD_KEYS = ["title", "name", "url", "link", "date", "createdAt", "publishedAt", "id"];

export function findArrayPathCandidates(data: unknown): JsonArrayPathCandidate[] {
  const results: JsonArrayPathCandidate[] = [];

  function walk(value: unknown, path: string) {
    if (Array.isArray(value)) {
      const firstObject = value.find(
        (item) => item && typeof item === "object" && !Array.isArray(item),
      ) as Record<string, unknown> | undefined;

      const sampleKeys = firstObject ? Object.keys(firstObject).slice(0, 20) : [];
      results.push({
        path,
        length: value.length,
        sampleKeys,
        confidence: scoreArrayCandidate(path, sampleKeys, value.length),
      });

      // Walk first 3 items to find nested arrays (but don't add numeric paths as candidates)
      value.slice(0, 3).forEach((item) => {
        if (item && typeof item === "object") {
          for (const [k, v] of Object.entries(item as Record<string, unknown>)) {
            walk(v, `${path}.${k}`);
          }
        }
      });
      return;
    }

    if (value && typeof value === "object") {
      for (const [key, child] of Object.entries(value as Record<string, unknown>)) {
        walk(child, path ? `${path}.${key}` : key);
      }
    }
  }

  walk(data, "");

  return results
    .filter((c) => c.length > 0)
    .sort((a, b) => b.confidence - a.confidence);
}

export function getNestedValue(data: unknown, path: string): unknown {
  const parts = path.split(".");
  let current = data;
  for (const part of parts) {
    if (current == null || typeof current !== "object") return undefined;
    current = (current as Record<string, unknown>)[part];
  }
  return current;
}

function scoreArrayCandidate(path: string, sampleKeys: string[], length: number): number {
  let score = 0.1;
  const pathLower = path.toLowerCase();

  for (const keyword of HIGH_CONFIDENCE_PATH_KEYWORDS) {
    if (pathLower.endsWith(keyword)) {
      score += 0.3;
      break;
    }
  }

  for (const key of sampleKeys) {
    if (HIGH_CONFIDENCE_FIELD_KEYS.includes(key.toLowerCase())) {
      score += 0.1;
    }
  }

  if (length > 1) score += 0.05;
  if (length > 5) score += 0.05;

  return Math.min(score, 1.0);
}
```

```ts
import { getNestedValue } from "./structured-feed.utility";
import type { StructuredFeedMapping } from "../models/graphql.model";
import type { FeedConfig } from "../models/feed-config.model";

export function buildRSSFromStructuredData(
  data: unknown,
  mapping: StructuredFeedMapping,
  feedConfig: FeedConfig,
): string {
  const items = getNestedValue(data, mapping.itemPath);
  if (!Array.isArray(items)) {
    throw new Error(
      `Structured item path "${mapping.itemPath}" did not resolve to an array (got ${items == null ? "null/undefined" : typeof items})`,
    );
  }

  const feedName = feedConfig.feedName ?? "Feed";

  const rssItems = items.map((item) => {
    const title = escapeXml(String(getNestedValue(item, mapping.title ?? "") ?? ""));
    const link = String(getNestedValue(item, mapping.link ?? "") ?? "");
    const description = String(getNestedValue(item, mapping.description ?? "") ?? "");
    const pubDate = (() => {
      const v = getNestedValue(item, mapping.pubDate ?? "");
      if (!v) return "";
      try { return new Date(String(v)).toUTCString(); } catch { return ""; }
    })();
    const guid = String(getNestedValue(item, mapping.guid ?? mapping.link ?? "") ?? "");
    const author = String(getNestedValue(item, mapping.author ?? "") ?? "");
    const categories = (() => {
      const v = getNestedValue(item, mapping.categories ?? "");
      if (!v) return "";
      const arr = Array.isArray(v) ? v : [v];
      return arr.map((c) => `<category>${escapeXml(String(c))}</category>`).join("");
    })();

    return `
    <item>
      ${title ? `<title>${title}</title>` : ""}
      ${link ? `<link>${escapeXml(link)}</link>` : ""}
      ${guid ? `<guid isPermaLink="false">${escapeXml(guid)}</guid>` : ""}
      ${description ? `<description><![CDATA[${description}]]></description>` : ""}
      ${pubDate ? `<pubDate>${pubDate}</pubDate>` : ""}
      ${author ? `<author>${escapeXml(author)}</author>` : ""}
      ${categories}
    </item>`;
  });

  return `<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0">
  <channel>
    <title>${escapeXml(feedName)}</title>
    <link></link>
    <description>${escapeXml(feedName)} feed</description>
    ${rssItems.join("")}
  </channel>
</rss>`;
}

function escapeXml(str: string): string {
  return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}
```

```ts
// utilities/graphql-feed.utility.ts
import type {
  ExecuteGraphQLFeedOptions,
  GraphQLFeedConfig,
  StructuredFeedSourceResult,
} from "../models/graphql.model";

export function resolveEnvPlaceholders(value: string): string {
  return value.replace(/\$\{([A-Z0-9_]+)\}/gi, (_, key) => {
    return process.env[key] ?? "";
  });
}

function resolveHeaderValues(headers: Record<string, string>): Record<string, string> {
  const resolved: Record<string, string> = {};
  for (const [key, value] of Object.entries(headers)) {
    resolved[key] = resolveEnvPlaceholders(value);
  }
  return resolved;
}

export function validateGraphQLConfig(input: unknown): GraphQLFeedConfig {
  if (!input || typeof input !== "object") throw new Error("GraphQL config must be an object");
  const cfg = input as Record<string, unknown>;

  if (!cfg.endpoint || typeof cfg.endpoint !== "string") {
    throw new Error("GraphQL config: endpoint is required");
  }
  const url = new URL(cfg.endpoint);
  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error(`GraphQL config: endpoint must use http or https, got "${url.protocol}"`);
  }

  if (!cfg.query || typeof cfg.query !== "string" || !cfg.query.trim()) {
    throw new Error("GraphQL config: query is required and must not be empty");
  }

  if (cfg.variables !== undefined) {
    if (typeof cfg.variables !== "object" || Array.isArray(cfg.variables) || cfg.variables === null) {
      throw new Error("GraphQL config: variables must be a JSON object");
    }
  }

  if (!cfg.mapping || typeof cfg.mapping !== "object") {
    throw new Error("GraphQL config: mapping is required");
  }
  const mapping = cfg.mapping as Record<string, unknown>;
  if (!mapping.itemPath || typeof mapping.itemPath !== "string") {
    throw new Error("GraphQL config: mapping.itemPath is required");
  }

  const timeoutMs = typeof cfg.timeoutMs === "number" ? cfg.timeoutMs : 15000;
  if (timeoutMs < 1000 || timeoutMs > 60000) {
    throw new Error("GraphQL config: timeoutMs must be between 1000 and 60000");
  }

  return {
    endpoint: cfg.endpoint,
    method: "POST",
    headers: cfg.headers as Record<string, string> | undefined,
    query: cfg.query,
    variables: cfg.variables as Record<string, unknown> | undefined,
    operationName: cfg.operationName as string | undefined,
    timeoutMs,
    mapping: cfg.mapping as GraphQLFeedConfig["mapping"],
    pagination: { enabled: false },
  };
}

export async function executeGraphQLFeedRequest(
  options: ExecuteGraphQLFeedOptions,
): Promise<StructuredFeedSourceResult> {
  const started = Date.now();
  const warnings: string[] = [];

  const resolvedHeaders = resolveHeaderValues({
    "Content-Type": "application/json",
    Accept: "application/json",
    ...(options.headers ?? {}),
  });

  let response: Response;
  try {
    response = await fetch(options.endpoint, {
      method: "POST",
      headers: resolvedHeaders,
      body: JSON.stringify({
        query: options.query,
        variables: options.variables ?? {},
        operationName: options.operationName,
      }),
      signal: AbortSignal.timeout(options.timeoutMs ?? 15000),
    });
  } catch (err) {
    throw new Error(`GraphQL request failed: ${String(err)}`);
  }

  if (response.status === 401 || response.status === 403) {
    throw new Error(`GraphQL request failed: authentication required (HTTP ${response.status})`);
  }
  if (response.status === 429) {
    throw new Error(`GraphQL request failed: rate limited (HTTP ${response.status})`);
  }
  if (!response.ok) {
    throw new Error(`GraphQL request failed: HTTP ${response.status}`);
  }

  let data: Record<string, unknown>;
  try {
    data = (await response.json()) as Record<string, unknown>;
  } catch {
    throw new Error("GraphQL response could not be parsed as JSON");
  }

  const responseBytes = JSON.stringify(data).length;
  const gqlErrors = Array.isArray(data.errors) ? data.errors : [];

  if (gqlErrors.length > 0) {
    if (!data.data) {
      const firstError = (gqlErrors[0] as Record<string, unknown>).message ?? "Unknown error";
      throw new Error(`GraphQL returned errors without data: ${firstError}`);
    }
    warnings.push(`GraphQL returned ${gqlErrors.length} error(s) alongside partial data`);
  }

  return {
    data,
    warnings,
    stats: { fetchMs: Date.now() - started, responseBytes },
  };
}
```

```tsx
// frontend/src/components/forms/GraphQLForm.tsx
import React, { useState } from "react";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { Badge } from "@/components/ui/badge";
import type { GraphQLFeedConfig, JsonArrayPathCandidate } from "@/models/graphql.model";

interface GraphQLFormProps {
  value: Partial<GraphQLFeedConfig>;
  onChange: (config: Partial<GraphQLFeedConfig>) => void;
}

export function GraphQLForm({ value, onChange }: GraphQLFormProps) {
  const [queryResult, setQueryResult] = useState<{
    success: boolean;
    candidates?: JsonArrayPathCandidate[];
    warnings?: string[];
    stats?: { fetchMs: number; responseBytes: number };
    error?: string;
  } | null>(null);
  const [running, setRunning] = useState(false);
  const [variablesError, setVariablesError] = useState<string | null>(null);

  const config: Partial<GraphQLFeedConfig> = {
    method: "POST",
    timeoutMs: 15000,
    mapping: { itemPath: "" },
    ...value,
  };

  function handleVariablesChange(raw: string) {
    if (!raw.trim()) {
      onChange({ ...config, variables: undefined });
      setVariablesError(null);
      return;
    }
    try {
      const parsed = JSON.parse(raw);
      if (typeof parsed !== "object" || Array.isArray(parsed) || parsed === null) {
        setVariablesError("Variables must be a JSON object");
        return;
      }
      setVariablesError(null);
      onChange({ ...config, variables: parsed });
    } catch {
      setVariablesError("Invalid JSON");
    }
  }

  async function handleRunQuery() {
    if (!config.endpoint || !config.query) return;
    setRunning(true);
    try {
      const res = await fetch("/preview/graphql", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ config }),
      });
      const data = await res.json();
      setQueryResult(data);
      if (data.success && data.arrayPathCandidates?.length > 0 && !config.mapping?.itemPath) {
        onChange({ ...config, mapping: { ...(config.mapping ?? { itemPath: "" }), itemPath: data.arrayPathCandidates[0].path } });
      }
    } finally {
      setRunning(false);
    }
  }

  const mapping = config.mapping ?? { itemPath: "" };

  return (
    <div className="space-y-6">
      {/* Endpoint */}
      <div className="space-y-2">
        <Label>GraphQL endpoint</Label>
        <Input
          placeholder="https://api.example.com/graphql"
          value={config.endpoint ?? ""}
          onChange={(e) => onChange({ ...config, endpoint: e.target.value })}
        />
      </div>

      {/* Headers */}
      <div className="space-y-2">
        <Label>Headers</Label>
        <Textarea
          placeholder={"Authorization: Bearer ${GITHUB_TOKEN}\nContent-Type: application/json"}
          rows={3}
          value={Object.entries(config.headers ?? {}).map(([k, v]) => `${k}: ${v}`).join("\n")}
          onChange={(e) => {
            const headers: Record<string, string> = {};
            for (const line of e.target.value.split("\n")) {
              const idx = line.indexOf(":");
              if (idx > 0) headers[line.slice(0, idx).trim()] = line.slice(idx + 1).trim();
            }
            onChange({ ...config, headers });
          }}
        />
        <p className="text-xs text-muted-foreground">Use $&#123;ENV_VAR&#125; for secrets stored in environment variables.</p>
      </div>

      {/* Query */}
      <div className="space-y-2">
        <Label>GraphQL query</Label>
        <Textarea
          placeholder={"query LatestPosts {\n  posts {\n    nodes {\n      title\n      url\n      publishedAt\n    }\n  }\n}"}
          rows={8}
          value={config.query ?? ""}
          onChange={(e) => onChange({ ...config, query: e.target.value })}
          className="font-mono text-sm"
        />
      </div>

      {/* Variables + Operation Name */}
      <div className="grid grid-cols-2 gap-4">
        <div className="space-y-2">
          <Label>Variables (JSON object)</Label>
          <Textarea
            placeholder={'{\n  "owner": "example"\n}'}
            rows={4}
            defaultValue={config.variables ? JSON.stringify(config.variables, null, 2) : ""}
            onChange={(e) => handleVariablesChange(e.target.value)}
            className="font-mono text-sm"
          />
          {variablesError && <p className="text-xs text-destructive">{variablesError}</p>}
        </div>
        <div className="space-y-2">
          <Label>Operation name (optional)</Label>
          <Input
            placeholder="LatestPosts"
            value={config.operationName ?? ""}
            onChange={(e) => onChange({ ...config, operationName: e.target.value || undefined })}
          />
          <Label className="mt-4 block">Timeout (ms)</Label>
          <Input
            type="number"
            value={config.timeoutMs ?? 15000}
            onChange={(e) => onChange({ ...config, timeoutMs: Number(e.target.value) })}
            min={1000}
            max={60000}
          />
        </div>
      </div>

      {/* Run Query */}
      <Button onClick={handleRunQuery} disabled={running || !config.endpoint || !config.query}>
        {running ? "Running..." : "Run Query"}
      </Button>

      {queryResult && (
        <div className="border rounded p-3 space-y-2 text-sm">
          {queryResult.success ? (
            <>
              <p className="text-green-600 font-medium">Query succeeded</p>
              {queryResult.stats && (
                <p className="text-muted-foreground">{queryResult.stats.fetchMs}ms · {(queryResult.stats.responseBytes / 1024).toFixed(1)} KB</p>
              )}
              {queryResult.candidates && queryResult.candidates.length > 0 && (
                <div>
                  <p className="font-medium">Detected arrays:</p>
                  <ul className="mt-1 space-y-1">
                    {queryResult.candidates.slice(0, 5).map((c) => (
                      <li key={c.path} className="flex items-center gap-2">
                        <button
                          className="text-primary underline font-mono text-xs"
                          onClick={() => onChange({ ...config, mapping: { ...mapping, itemPath: c.path } })}
                        >
                          {c.path}
                        </button>
                        <Badge variant="secondary">{c.length} items</Badge>
                      </li>
                    ))}
                  </ul>
                </div>
              )}
              {queryResult.warnings?.map((w, i) => (
                <p key={i} className="text-yellow-600 text-xs">{w}</p>
              ))}
            </>
          ) : (
            <p className="text-destructive">{queryResult.error}</p>
          )}
        </div>
      )}

      {/* Mapping */}
      <div className="space-y-3">
        <Label className="text-base font-medium">Field mapping</Label>
        {[
          { key: "itemPath", label: "Item path", required: true },
          { key: "title", label: "Title" },
          { key: "link", label: "Link" },
          { key: "description", label: "Description" },
          { key: "pubDate", label: "Date" },
          { key: "guid", label: "GUID" },
          { key: "author", label: "Author" },
          { key: "categories", label: "Categories" },
        ].map(({ key, label, required }) => (
          <div key={key} className="grid grid-cols-3 items-center gap-4">
            <Label className={required ? "font-medium" : ""}>{label}{required ? " *" : ""}</Label>
            <Input
              className="col-span-2 font-mono text-sm"
              placeholder={key === "itemPath" ? "data.posts.nodes" : `e.g. ${key}`}
              value={(mapping as Record<string, string | undefined>)[key] ?? ""}
              onChange={(e) => onChange({ ...config, mapping: { ...mapping, [key]: e.target.value || undefined } })}
            />
          </div>
        ))}
      </div>
    </div>
  );
}
```
