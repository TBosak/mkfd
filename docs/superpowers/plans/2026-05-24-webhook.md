**Goal:** Add `feedType: webhook` so Mkfd can receive incoming HTTP events and turn them into RSS feeds, with per-feed bearer token auth, JSONL event storage, deduplication, rate limiting, and immediate feed regeneration on receipt.

**Architecture:** New `models/webhook.model.ts` defines all types; `utilities/webhook-feed.utility.ts` owns token generation/hashing/verification, payload validation, normalization, JSONL storage, and pruning; `utilities/rss-builder.utility.ts` gains `buildRSSFromWebhookEvents`; `routes/webhook.ts` exposes `POST /webhook-feeds/:slug` with in-memory rate limiting; `frontend/src/components/forms/WebhookFeedForm.tsx` provides the builder UI with auto-generated curl example.

**Tech Stack:** Bun, TypeScript, Hono, React 18, shadcn/ui, bun:test, Node.js `crypto` for SHA-256 and random bytes

```ts
// utilities/webhook-feed.utility.ts
import { createHash, randomBytes, timingSafeEqual } from "crypto";

export function generateWebhookToken(): string {
  return "mkfd_wh_" + randomBytes(32).toString("hex");
}

export function hashWebhookToken(token: string): string {
  return createHash("sha256").update(token).digest("hex");
}

export function verifyWebhookToken(token: string, tokenHash: string): boolean {
  if (!token) return false;
  const supplied = Buffer.from(hashWebhookToken(token), "hex");
  const stored = Buffer.from(tokenHash, "hex");
  if (supplied.length !== stored.length) return false;
  return timingSafeEqual(supplied, stored);
}
```

```ts
// append to utilities/webhook-feed.utility.ts
import { randomUUID } from "crypto";
import type { WebhookFeedPayload, WebhookFeedEvent } from "../models/webhook.model";

export function validateWebhookPayload(payload: unknown): WebhookFeedPayload {
  if (typeof payload !== "object" || payload === null) {
    throw new Error("Payload must be a JSON object");
  }
  const p = payload as Record<string, unknown>;

  if (!("title" in p) || p.title === undefined) {
    throw new Error("title is required");
  }
  if (typeof p.title !== "string") {
    throw new Error("title must be a string");
  }
  if (p.title.length > 300) {
    throw new Error("title must be 300 characters or fewer");
  }

  if (p.description !== undefined) {
    if (typeof p.description !== "string") throw new Error("description must be a string");
    if (p.description.length > 20000) throw new Error("description must be 20000 characters or fewer");
  }

  if (p.url !== undefined) {
    if (typeof p.url !== "string") throw new Error("url must be a string");
    try {
      const u = new URL(p.url);
      if (u.protocol !== "http:" && u.protocol !== "https:") {
        throw new Error("url must use http or https");
      }
    } catch {
      throw new Error("url must be a valid http or https URL");
    }
  }

  if (p.date !== undefined && typeof p.date !== "string") {
    throw new Error("date must be a string");
  }

  if (p.categories !== undefined) {
    if (!Array.isArray(p.categories)) throw new Error("categories must be an array");
    if (p.categories.length > 25) throw new Error("categories must have 25 or fewer items");
    for (const cat of p.categories) {
      if (typeof cat !== "string") throw new Error("each category must be a string");
      if (cat.length > 100) throw new Error("each category must be 100 characters or fewer");
    }
  }

  const validSeverities = ["info", "success", "warning", "error"];
  if (p.severity !== undefined) {
    if (!validSeverities.includes(p.severity as string)) {
      throw new Error(`severity must be one of: ${validSeverities.join(", ")}`);
    }
  }

  if (p.metadata !== undefined) {
    if (typeof p.metadata !== "object" || Array.isArray(p.metadata) || p.metadata === null) {
      throw new Error("metadata must be an object");
    }
    const serialized = JSON.stringify(p.metadata);
    if (serialized.length > 10240) throw new Error("metadata must be 10 KB or smaller");
  }

  return p as unknown as WebhookFeedPayload;
}

export function normalizeWebhookEvent(
  feedId: string,
  payload: WebhookFeedPayload,
  receivedAt: Date,
): WebhookFeedEvent {
  const id = randomUUID();
  const receivedAtIso = receivedAt.toISOString();

  let eventDate = receivedAtIso;
  if (payload.date) {
    const parsed = new Date(payload.date);
    if (!isNaN(parsed.getTime())) {
      eventDate = parsed.toISOString();
    }
  }

  let dedupeKey: string;
  if (payload.id) {
    dedupeKey = payload.id;
  } else {
    const raw = payload.title + (payload.url ?? "") + eventDate;
    dedupeKey = createHash("sha256").update(raw).digest("hex");
  }

  return {
    id,
    feedId,
    externalId: payload.id,
    receivedAt: receivedAtIso,
    eventDate,
    title: payload.title,
    description: payload.description,
    link: payload.url,
    author: payload.author,
    categories: payload.categories ?? [],
    severity: payload.severity,
    metadata: payload.metadata,
    dedupeKey,
  };
}
```

```ts
// append to utilities/webhook-feed.utility.ts
import { existsSync, readFileSync, writeFileSync, mkdirSync } from "fs";
import { join } from "path";
import type { WebhookFeedConfig, WebhookFeedEvent } from "../models/webhook.model";

const WEBHOOK_EVENTS_DIR = "./feed-state/webhook-events";

function ensureDir() {
  if (!existsSync(WEBHOOK_EVENTS_DIR)) {
    mkdirSync(WEBHOOK_EVENTS_DIR, { recursive: true });
  }
}

function eventFilePath(feedId: string): string {
  return join(WEBHOOK_EVENTS_DIR, `${feedId}.jsonl`);
}

async function readAllEvents(feedId: string): Promise<WebhookFeedEvent[]> {
  const path = eventFilePath(feedId);
  if (!existsSync(path)) return [];
  const lines = readFileSync(path, "utf-8").split("\n").filter(Boolean);
  const events: WebhookFeedEvent[] = [];
  for (const line of lines) {
    try {
      events.push(JSON.parse(line) as WebhookFeedEvent);
    } catch {
      console.warn(`[webhook] Skipping corrupt JSONL line in ${feedId}`);
    }
  }
  return events;
}

export async function appendWebhookEvent(
  feedId: string,
  event: WebhookFeedEvent,
  _config: WebhookFeedConfig,
): Promise<{ inserted: boolean; duplicate: boolean }> {
  ensureDir();
  const existing = await readAllEvents(feedId);
  const isDuplicate = existing.some((e) => e.dedupeKey === event.dedupeKey);
  if (isDuplicate) return { inserted: false, duplicate: true };

  const path = eventFilePath(feedId);
  const line = JSON.stringify(event) + "\n";
  await Bun.write(path, (existsSync(path) ? readFileSync(path, "utf-8") : "") + line);
  return { inserted: true, duplicate: false };
}

export async function loadWebhookEvents(
  feedId: string,
  limit: number,
): Promise<WebhookFeedEvent[]> {
  const events = await readAllEvents(feedId);
  return events
    .sort((a, b) => new Date(b.eventDate).getTime() - new Date(a.eventDate).getTime())
    .slice(0, limit);
}

export async function pruneWebhookEvents(
  feedId: string,
  retentionDays: number,
  maxItems: number,
): Promise<void> {
  const events = await readAllEvents(feedId);
  const cutoff = new Date();
  cutoff.setDate(cutoff.getDate() - retentionDays);
  const cutoffMs = cutoff.getTime();

  const retained = events
    .filter((e) => new Date(e.eventDate).getTime() >= cutoffMs)
    .sort((a, b) => new Date(b.eventDate).getTime() - new Date(a.eventDate).getTime())
    .slice(0, maxItems);

  ensureDir();
  writeFileSync(
    eventFilePath(feedId),
    retained.map((e) => JSON.stringify(e)).join("\n") + (retained.length ? "\n" : ""),
    "utf-8",
  );
}
```

```ts
import type { WebhookFeedEvent } from "../models/webhook.model";

function xmlEscape(s: string): string {
  return s
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;");
}

export function buildRSSFromWebhookEvents(
  events: WebhookFeedEvent[],
  feedConfig: FeedConfig,
): string {
  const feedTitle = xmlEscape(feedConfig.feedName ?? "Webhook Feed");
  const now = new Date().toUTCString();
  const items = events
    .map((e) => {
      const guid = e.externalId ?? e.id;
      const categories = [
        ...e.categories,
        ...(e.severity ? [e.severity] : []),
      ]
        .map((c) => `<category>${xmlEscape(c)}</category>`)
        .join("\n        ");

      return `  <item>
      <title>${xmlEscape(e.title)}</title>
      ${e.link ? `<link>${xmlEscape(e.link)}</link>` : ""}
      <guid>${xmlEscape(guid)}</guid>
      <pubDate>${new Date(e.eventDate).toUTCString()}</pubDate>
      ${e.author ? `<author>${xmlEscape(e.author)}</author>` : ""}
      ${e.description ? `<description>${xmlEscape(e.description)}</description>` : ""}
      ${categories}
    </item>`;
    })
    .join("\n");

  return `<?xml version="1.0" encoding="UTF-8"?>
<rss version="2.0">
  <channel>
    <title>${feedTitle}</title>
    <description>${feedTitle}</description>
    <lastBuildDate>${now}</lastBuildDate>
    ${items}
  </channel>
</rss>`;
}
```

```ts
// routes/webhook.ts
import { Hono } from "hono";
import {
  verifyWebhookToken,
  validateWebhookPayload,
  normalizeWebhookEvent,
  appendWebhookEvent,
  pruneWebhookEvents,
  loadWebhookEvents,
} from "../utilities/webhook-feed.utility";
import { buildRSSFromWebhookEvents } from "../utilities/rss-builder.utility";
import type { FeedConfig } from "../models/feed-config.model";

// In-memory rate limiter: 60 req/min per slug
const rateLimits = new Map<string, { count: number; resetAt: number }>();

function checkRateLimit(slug: string): boolean {
  const now = Date.now();
  const entry = rateLimits.get(slug);
  if (!entry || entry.resetAt <= now) {
    rateLimits.set(slug, { count: 1, resetAt: now + 60_000 });
    return true;
  }
  if (entry.count >= 60) return false;
  entry.count++;
  return true;
}

export function createWebhookRoutes(
  app: Hono,
  getFeeds: () => FeedConfig[],
  writeFeedXml: (feedName: string, xml: string) => Promise<void>,
) {
  app.post("/webhook-feeds/:slug", async (c) => {
    const slug = c.req.param("slug");

    // Step 1: Locate feed config
    const feed = getFeeds().find(
      (f) => f.feedType === "webhook" && f.webhookFeed?.slug === slug,
    );
    if (!feed || !feed.webhookFeed) {
      return c.json({ ok: false, error: "Feed not found" }, 404);
    }

    // Step 2: Check Authorization header
    const authHeader = c.req.header("Authorization");
    if (!authHeader?.startsWith("Bearer ")) {
      return c.json({ ok: false, error: "Missing Authorization bearer token" }, 401);
    }
    const token = authHeader.slice("Bearer ".length).trim();

    // Step 3: Verify token
    if (!verifyWebhookToken(token, feed.webhookFeed.tokenHash)) {
      return c.json({ ok: false, error: "Invalid token" }, 403);
    }

    // Step 4: Check Content-Type
    const contentType = c.req.header("Content-Type") ?? "";
    if (!contentType.includes("application/json")) {
      return c.json({ ok: false, error: "Content-Type must be application/json" }, 415);
    }

    // Step 5: Body size check (64 KB)
    const rawBody = await c.req.text();
    if (rawBody.length > 65536) {
      return c.json({ ok: false, error: "Payload too large (max 64 KB)" }, 413);
    }

    // Step 6: Parse JSON
    let body: unknown;
    try {
      body = JSON.parse(rawBody);
    } catch {
      return c.json({ ok: false, error: "Invalid JSON" }, 400);
    }

    // Step 7: Validate payload
    let payload;
    try {
      payload = validateWebhookPayload(body);
    } catch (err: unknown) {
      const message = err instanceof Error ? err.message : "Invalid payload";
      return c.json({ ok: false, error: message }, 400);
    }

    // Step 8: Rate limit
    if (!checkRateLimit(slug)) {
      return c.json({ ok: false, error: "Rate limit exceeded (60/min)" }, 429);
    }

    // Step 9: Normalize event
    const feedId = feed.feedId;
    const event = normalizeWebhookEvent(feedId, payload, new Date());

    // Step 10: Append (deduplicate)
    const result = await appendWebhookEvent(feedId, event, feed.webhookFeed);
    if (result.duplicate) {
      return c.json({ ok: true, duplicate: true }, 200);
    }

    // Step 11: Prune
    await pruneWebhookEvents(
      feedId,
      feed.webhookFeed.retentionDays,
      feed.webhookFeed.maxItems,
    );

    // Step 12: Rebuild and write feed XML
    const events = await loadWebhookEvents(feedId, feed.webhookFeed.maxItems);
    const xml = buildRSSFromWebhookEvents(events, feed);
    await writeFeedXml(feed.feedName, xml);

    // Step 13: Return success
    const feedUrl = `/public/feeds/${feed.feedName}.xml`;
    return c.json({ ok: true, eventId: event.id, feedUrl }, 200);
  });
}
```
